{{- /* Resolve a gemoji key (an ingredient id, an ingredient's `emoji` field, or a timer's emoji key) to a glyph: data/ingredient_emoji.yaml override wins, else Hugo's `emojify`, else empty string. */ -}} {{- $key := .key -}} {{- $emoji := "" -}} {{- with index site.Data.ingredient_emoji $key -}} {{- $emoji = . -}} {{- else -}} {{- $code := printf ":%s:" $key -}} {{- $try := emojify $code -}} {{- if ne $try $code -}}{{- $emoji = $try -}}{{- end -}} {{- end -}} {{- return $emoji -}}
